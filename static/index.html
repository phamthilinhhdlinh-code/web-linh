Cấu Trúc Bảng CSDL (Relational SQL Schema) 

-- 1. Bảng Lớp Học & Nhóm Thi Đua
CREATE TABLE classes (id INTEGER PRIMARY KEY, name TEXT, grade_level INT, academic_year TEXT);
CREATE TABLE groups (id INTEGER PRIMARY KEY, class_id INT REFERENCES classes(id), group_number INT, name TEXT);

-- 2. Bảng Học Sinh
CREATE TABLE students (id INTEGER PRIMARY KEY, student_code TEXT UNIQUE, full_name TEXT, group_id INT REFERENCES groups(id), is_group_leader INT);

-- 3. Bảng Điểm KTTX (Kiểm Tra Thường Xuyên)
CREATE TABLE regular_scores (id INTEGER PRIMARY KEY, student_id INT REFERENCES students(id), score_type_id INT, score REAL, date_logged TEXT);

-- 4. Bảng Nhật Ký Khai Báo Điểm Cộng/Trừ từ Nhóm Trưởng
CREATE TABLE bonus_penalty_logs (
    id INTEGER PRIMARY KEY, student_id INT, group_id INT, type TEXT CHECK(type IN ('BONUS', 'PENALTY')),
    points REAL, reason TEXT, category_type TEXT, declared_by_student_id INT,
    status TEXT CHECK(status IN ('PENDING', 'APPROVED', 'REJECTED')), created_at TEXT
);

-- 5. Bảng Điểm Chốt Môn KHTN
CREATE TABLE final_grades (
    student_id INT UNIQUE REFERENCES students(id), avg_kttx REAL, total_bonus_penalty REAL,
    final_score REAL, academic_rank TEXT, updated_at TEXT
);
